      group = fish_id

    alpha = 0.12,
    linewidth = 0.4,
    colour = "grey40"
  ) +

  # Raw observations
  geom_point(
    data = mass_long,
    aes(
      x = time,
      y = body_mass,
      colour = treatment_plot
    ),
    alpha = 0.35,
    size = 1.8,
    position = position_jitter(width = 0.05, height = 0)
  ) +

  # Estimated marginal means
  geom_line(
    data = emm_mass_plot_df,
    aes(
      x = time,
      y = emmean,
      group = treatment_plot,
      colour = treatment_plot
    ),
    linewidth = 1.4
  ) +

  # 95% CI
  geom_errorbar(
    data = emm_mass_plot_df,
    aes(
      x = time,
      ymin = lower.CL,
      ymax = upper.CL,
      colour = treatment_plot
    ),
    width = 0.08,
    linewidth = 1.1
  ) +

  geom_point(

    stroke = 1.1,
    colour = "black"

  facet_wrap(~ treatment_plot) +

  scale_colour_manual(values = dark_cols) +

  scale_y_continuous(

  labs(
